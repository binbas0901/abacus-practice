The abacus

Wooden frame, red heaven beads (worth 5), yellow earth beads (worth 1), white dots on the beam marking the ones rod. Beads slide when the number changes.

How the app plugs it in

<link rel="stylesheet" href="look/tokens.css">
<script src="look/characters.js"></script>

<ab-juni pose="point" size="160"></ab-juni>
<ab-bixi pose="confused" size="120" idle></ab-bixi>

<script src="look/abacus.js"></script>
<ab-abacus value="7" rods="5" labels highlight="1 e1 up"></ab-abacus>

// in code
await abacus.setValue(10);                 // beads slide
await abacus.setRod(0, {heaven:false});    // move one rod
abacus.highlight(1, 'e1', 'up');           // ring + arrow
abacus.addEventListener('change', e => e.detail.value);

// in code: change a pose
document.querySelector('ab-bixi').pose = 'cheer';
